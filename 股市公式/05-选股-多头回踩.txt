{==================================================
  多头回踩  （类型：条件选股公式）
  条件：均线多头排列（5>10>20>60）且 20 日线向上，
        当日最低价回踩到 10 日线附近但收在 10 日线之上，并且缩量
==================================================}

MA5:=MA(C,5);
MA10:=MA(C,10);
MA20:=MA(C,20);
MA60:=MA(C,60);
XG:MA5>MA10 AND MA10>MA20 AND MA20>MA60
   AND MA20>REF(MA20,1)
   AND L<=MA10*1.01 AND C>MA10
   AND V<MA(V,5)
   AND BARSCOUNT(C)>120
   AND NOT(NAMELIKE('ST')) AND NOT(NAMELIKE('*ST'));
