{==================================================
  强于大盘  （类型：条件选股公式）
  条件：近20日比大盘多涨10%以上，且站在20日线上、20日线向上
==================================================}

RS:=C/INDEXC;
MA20:=MA(C,20);
XG:RS/REF(RS,20)>1.10
   AND C>MA20 AND MA20>REF(MA20,1)
   AND BARSCOUNT(C)>120
   AND NOT(NAMELIKE('ST')) AND NOT(NAMELIKE('*ST'));
