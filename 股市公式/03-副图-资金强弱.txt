{==================================================
  资金强弱  （类型：技术指标公式 → 副图）
  作用：按“收盘价在当日振幅中的位置 × 成交量”估算买卖力量
        （类似 Chaikin 资金流 CMF），红柱=资金偏流入，绿柱=偏流出
  另画一条“相对大盘”线：近20日跑赢大盘的幅度（%）
==================================================}

X:=(2*C-H-L)/(H-L+0.001)*V;
资金:EMA(SUM(X,20)/SUM(V,20)*100,3),COLORWHITE,LINETHICK2;
STICKLINE(资金>0,0,资金,2,0),COLORRED;
STICKLINE(资金<=0,0,资金,2,0),COLORGREEN;

RS:=C/INDEXC;
相对大盘:(RS/REF(RS,20)-1)*100,COLORYELLOW;
零轴:0,COLORGRAY;
